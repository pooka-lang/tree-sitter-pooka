with_propagation(value: Result(Int, String)) -> Result(Int, String) {
  let number = value?
  Ok(number + 1)
}
without_propagation(value: Result(Int, String)) -> Result(Int, String) {
  match value {
    Ok(number) -> Ok(number + 1)
    Error(error) -> Error(error)
  }
}
with_assert(value: Result(Int, String)) -> Int {
  let assert Ok(number) = value as "expected success"
  number
}
without_assert(value: Result(Int, String)) -> Int {
  match value {
    Ok(number) -> number
    Error(_) -> panic as "expected success"
  }
}
